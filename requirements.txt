#   conda install -c conda-forge gdal python-pdal
# experiments/sentinel_lab.py also needs rasterio; install it from conda-forge
# too (the pip wheel bundles its own GDAL, which can clash with conda's):
#   conda install -c conda-forge rasterio

# Earthwork model training + scan (earthwork_model/, earthwork_inference/) need
# PyTorch.  Install it from conda-forge into the SAME env (it then shares the
# env's MKL/OpenMP; installed in pdal_env as pytorch 2.10 CPU on 2026-10-27):
#   conda install -n pdal_env -c conda-forge --override-channels pytorch torchvision
# conda-forge picks a CUDA build when an NVIDIA driver is present; training on
# CPU runs ~1.5 min per step, so use a GPU machine for it.  Do NOT pip-install
# torch into the conda env: its bundled OpenMP clashes with conda's MKL.
# "DLL load failed ... procedure could not be found" on `import torch` after
# PDAL means another conda install's Library\bin is on PATH (see lidar/__init__.py).
